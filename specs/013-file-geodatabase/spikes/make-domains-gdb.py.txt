from osgeo import gdal, ogr, osr
import shutil, os
gdal.UseExceptions()
out = 'dom.gdb'
if os.path.exists(out):
    shutil.rmtree(out)
ds = gdal.GetDriverByName('OpenFileGDB').Create(out, 0, 0, 0, gdal.GDT_Unknown)
ds.AddFieldDomain(ogr.CreateCodedFieldDomain('RoadClass', 'Road class', ogr.OFTInteger, ogr.OFSTNone, {1: 'Freeway', 2: 'Arterial', 3: 'Local'}))
ds.AddFieldDomain(ogr.CreateRangeFieldDomain('Lanes', 'Lane count', ogr.OFTInteger, ogr.OFSTNone, 1, True, 12, True))
srs = osr.SpatialReference(); srs.ImportFromEPSG(26912)
lyr = ds.CreateLayer('roads', srs, ogr.wkbLineString)
f = ogr.FieldDefn('class', ogr.OFTInteger); f.SetDomainName('RoadClass'); lyr.CreateField(f)
f = ogr.FieldDefn('lanes', ogr.OFTInteger); f.SetDomainName('Lanes'); lyr.CreateField(f)
lyr.CreateField(ogr.FieldDefn('name', ogr.OFTString))
for i, (c, l, n) in enumerate([(1, 6, 'I-15'), (2, 4, 'State St'), (3, 2, 'Elm')]):
    ft = ogr.Feature(lyr.GetLayerDefn()); ft.SetField('class', c); ft.SetField('lanes', l); ft.SetField('name', n)
    ft.SetGeometry(ogr.CreateGeometryFromWkt('LINESTRING(430000 4500000, 431000 4500%d00)' % (i + 1)))
    lyr.CreateFeature(ft)
ds = None
print('ok')
