<!doctype html><input id="f" type="file" multiple><pre id="out"></pre>
<script>
window.run = () => new Promise(async (res) => {
  const w = new Worker('/worker.js')
  const files = []
  for (const f of document.getElementById('f').files) files.push({ name: f.name, buf: await f.arrayBuffer() })
  w.onmessage = (e) => res(e.data)
  w.postMessage({ files })
})
</script>
