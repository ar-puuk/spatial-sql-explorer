// 2. GeoPackage via sql.js: the minimum tables OGC 12-128r18 requires.
function wkb(g) {
  const parts = []
  const u8 = (v) => parts.push(Uint8Array.of(v))
  const u32 = (v) => { const b = new DataView(new ArrayBuffer(4)); b.setUint32(0, v, true); parts.push(new Uint8Array(b.buffer)) }
  const f64 = (v) => { const b = new DataView(new ArrayBuffer(8)); b.setFloat64(0, v, true); parts.push(new Uint8Array(b.buffer)) }
  const ring = (r) => { u32(r.length); for (const [x, y] of r) { f64(x); f64(y) } }
  const poly = (p) => { u8(1); u32(3); u32(p.length); p.forEach(ring) }
  if (g.type === 'Polygon') poly(g.coordinates)
  else if (g.type === 'MultiPolygon') { u8(1); u32(6); u32(g.coordinates.length); g.coordinates.forEach(poly) }
  else if (g.type === 'Point') { u8(1); u32(1); f64(g.coordinates[0]); f64(g.coordinates[1]) }
  const n = parts.reduce((s, p) => s + p.length, 0)
  const out = new Uint8Array(n); let o = 0
  for (const p of parts) { out.set(p, o); o += p.length }
  return out
}
function bbox(g) {
  const xs = [], ys = []
  const walk = (c) => (typeof c[0] === 'number' ? (xs.push(c[0]), ys.push(c[1])) : c.forEach(walk))
  walk(g.coordinates)
  return [Math.min(...xs), Math.max(...xs), Math.min(...ys), Math.max(...ys)]
}
function gpkgBlob(g, srs) {
  const body = wkb(g)
  const head = new DataView(new ArrayBuffer(8 + 32))
  head.setUint8(0, 0x47); head.setUint8(1, 0x50); head.setUint8(2, 0); head.setUint8(3, 0b0000_0011) // little endian, xy envelope
  head.setInt32(4, srs, true)
  bbox(g).forEach((v, i) => head.setFloat64(8 + i * 8, v, true))
  const out = new Uint8Array(40 + body.length); out.set(new Uint8Array(head.buffer)); out.set(body, 40)
  return out
}
