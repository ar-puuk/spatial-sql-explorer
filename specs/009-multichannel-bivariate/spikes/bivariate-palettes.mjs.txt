// Bivariate 3x3 palettes by bilinear interpolation in CIELAB between four
// corners: low/low (light neutral), high x, high y, high/high. Scored by the
// smallest CIE76 difference between any two cells, normal and deuteranope.
const hex = (h) => [1, 3, 5].map((i) => parseInt(h.slice(i, i + 2), 16))
const lin = (v) => { v /= 255; return v <= 0.04045 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4 }
const unlin = (v) => 255 * (v <= 0.0031308 ? 12.92 * v : 1.055 * Math.max(v, 0) ** (1 / 2.4) - 0.055)
const W = [0.95047, 1, 1.08883]
const f = (t) => (t > 0.008856 ? Math.cbrt(t) : 7.787 * t + 16 / 116)
const fi = (t) => (t ** 3 > 0.008856 ? t ** 3 : (t - 16 / 116) / 7.787)
function toLab(c) { const [r, g, b] = c.map(lin); const x = (0.4124 * r + 0.3576 * g + 0.1805 * b) / W[0], y = 0.2126 * r + 0.7152 * g + 0.0722 * b, z = (0.0193 * r + 0.1192 * g + 0.9505 * b) / W[2]; return [116 * f(y) - 16, 500 * (f(x) - f(y)), 200 * (f(y) - f(z))] }
function fromLab([L, a, bb]) { const fy = (L + 16) / 116, fx = fy + a / 500, fz = fy - bb / 200; const x = fi(fx) * W[0], y = fi(fy), z = fi(fz) * W[2]; const r = 3.2406 * x - 1.5372 * y - 0.4986 * z, g = -0.9689 * x + 1.8758 * y + 0.0415 * z, b = 0.0557 * x - 0.204 * y + 1.057 * z; return [r, g, b].map((v) => Math.min(255, Math.max(0, unlin(v)))) }
const toHex = (c) => '#' + c.map((v) => Math.round(v).toString(16).padStart(2, '0')).join('')
const DEU = [[0.367322, 0.860646, -0.227968], [0.280085, 0.672501, 0.047413], [-0.01182, 0.04294, 0.968881]]
const deu = (c) => { const l = c.map(lin); return DEU.map((row) => unlin(Math.min(1, Math.max(0, row[0] * l[0] + row[1] * l[1] + row[2] * l[2])))) }
const dE = (a, b) => Math.hypot(...toLab(a).map((v, i) => v - toLab(b)[i]))
const minDE = (cells, t = (x) => x) => { let m = 1e9; for (let i = 0; i < 9; i++) for (let j = i + 1; j < 9; j++) m = Math.min(m, dE(t(cells[i]), t(cells[j]))); return m }
function grid(ll, hx, hy, hh) {
  const [A, B, C, D] = [ll, hx, hy, hh].map((h) => toLab(hex(h)))
  const cells = []
  for (let j = 0; j < 3; j++) for (let i = 0; i < 3; i++) {
    const u = i / 2, v = j / 2
    cells.push(fromLab([0, 1, 2].map((k) => (1 - u) * (1 - v) * A[k] + u * (1 - v) * B[k] + (1 - u) * v * C[k] + u * v * D[k])))
  }
  return cells
}
